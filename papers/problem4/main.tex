\begin{abstract}
Let $S$ be a reachable image of a strongly connected synchronizing automaton
and $\varnothing\ne A\subsetneq S$. We determine the maximum shortest
avoid-or-compress length over binary and arbitrary finite alphabets for
$m=n-|S|=4$ and $m=5$. The exact values are $\min(n-|A|,11)$ and
$\min(n-|A|,14)$, respectively. Uniform binary constructions attain both
bounds. For the upper bounds, short-pair graphs and actual-image constraints
reduce all parameters to finite necessary domains. Exhaustive checks exclude
the domains, apart from a single relaxation eliminated by a multipartite
overlap lemma. The largest calculation covers 3,206,988 domains.
\end{abstract}

\section{Problem and results}
All words act on the right and length counts \emph{original alphabet letters}.
A complete deterministic automaton is a triple $\mathcal A=(Q,\Sigma,\delta)$,
where $Q$ and $\Sigma$ are finite and $\delta$ is total.
It is strongly connected if every state reaches every other state, and
synchronizing if $|Qw|=1$ for some word $w$.
Let $n=|Q|$, let $S=Qv$ be an actual reachable image with $|S|=s$,
and let $\varnothing\ne A\subsetneq S$, $|A|=r$.
The admissible parameters are $n\ge2$ and $1\le r<s\le n$.
Put
\[
 \ac(S,A)=\min\{|w|:A\nsubseteq Sw\ \text{or}\ |Sw|<s\},
 \qquad c(S)=\min\{|w|:|Sw|<s\}.
\]
Thus avoidance means omitting \emph{at least one} protected state, and
$\ac(S,A)\le c(S)$. Let $\Max(n,s,r)$ maximize $\ac(S,A)$ over
strongly connected synchronizing automata with arbitrary finite alphabets;
$\Max_{\rm bin}$ restricts to binary alphabets. Write $m=n-s$ and $q=n-r$.
The avoid-or-compress lemma~\cite{szykula2018,szykula2026} and
the Frankl--Pin compression bound~\cite{frankl1982,pin1983} give
\begin{equation}\label{eq:universal}
 \Max_{\rm bin}(n,s,r)\le\Max(n,s,r)
 \le\min\{q,\tbinom{m+2}{2}\}.
\end{equation}
A proof of the $q$ bound appears in Appendix~\ref{sec:appendix}.
Open Problem~4 of the survey~\cite{szykula2026} asks for tight bounds
depending on the set cardinalities. The two-state image case is already
illustrated there, and related fixed-state avoiding-or-compressing sharpness
is discussed in~\cite{ferens2021}. Our main contribution concerns entire
dense-image parameter families.

\begin{theorem}[Dense-image extrema; computer-assisted upper bounds]\label{thm:main}
For all admissible parameters,
\[
 \boxed{\Max_{\rm bin}(n,n-4,r)=\Max(n,n-4,r)=\min(q,11)}
\]
and
\[
 \boxed{\Max_{\rm bin}(n,n-5,r)=\Max(n,n-5,r)=\min(q,14).}
\]
In every strongly connected synchronizing automaton, every $(n-4)$-subset
has $c(S)\le11$ when $n\ge11$, and every $(n-5)$-subset has $c(S)\le14$
when $n\ge16$. These pure-compression assertions do not require reachability.
\end{theorem}

The lower bounds are established by explicit binary automata. For the
upper bounds, Sections~\ref{sec:method} and~\ref{sec:computation} present
the short-pair reductions, exhaustive finite-domain coverage, and
computational basis. The mathematical supplement is
\texttt{PROOF.md}~\cite{repo}. The universal $q$ bound handles the
parameters below the pure-compression thresholds.
The codimension-three bound $C_3\le9$, used in both upper proofs,
has a paper proof in the supplement.

\begin{proposition}[Binary sharpness below the two-block threshold]\label{prop:smallq}
Let $m\ge4$ and $m+1\le q\le2m+1$. For every admissible $r$ there is a binary
strongly connected synchronizing automaton with a reachable $(n-m)$-image
$S$ and an $r$-set $A\subsetneq S$ such that $\ac(S,A)=q$.
\end{proposition}
\begin{proof}
Appendix~\ref{sec:appendix} supplies all three required constructions,
including their reaching and reset words.
The first cardinality face covers $q=m+1$; the second covers $q=m+2$.
For $m+3\le q\le2m+1$, take $g=q-m-1$ and $M=q-1$ in
Proposition~\ref{prop:gaps}. Then $g\ge2$, $M\ge2g$,
$n=r+M+1$, $s=r+g+1=n-m$, and $\ac(S,A)=M+1=q$.
\end{proof}

\subsection{Uniform construction and proof}
\begin{theorem}\label{thm:two-block}
Put $m=n-s\ge3$ and $q=n-r\ge2m+2$.
There is a binary strongly connected synchronizing automaton with reachable
$S$ attaining $\ac(S,A)=\min\{q,3m-1\}$.
Consequently $\Max_{\rm bin}=\Max=q$ whenever $2m+2\le q\le3m-1$.
For larger $q$ the construction gives only a lower bound.
\end{theorem}
\begin{proof}
Write $t=m-1\ge2$ and $p=n-2t-1\ge r+3$.
Take $P=\{p_0,\ldots,p_{p-1}\}$, $X=\{x_1,\ldots,x_t\}$,
$Y=\{y_1,\ldots,y_t\}$ and one further state $z$.
Letter $a$ interchanges the last two $P$-states, fixes the others, and
cycles $x_1,\ldots,x_t,y_1,\ldots,y_t,z$.
Letter $b$ sends $p_0$ to $y_1$, $p_i$ to $p_{i-1}$ for $i\ge1$,
increments indices in $X,Y$ below $t$, sends both $x_t,y_t$ to $p_{p-1}$,
and fixes $z$. Put $S=P\cup X$, $A=\{p_0,\ldots,p_{r-1}\}$ and $L=3t+2$.

\emph{No compression before $L$.}
For $\rho_i=\min\{i,p-2\}$ define a symmetric potential $\Phi$ on unordered
pairs, zero on singleton pairs, by the following table. Indices in $X,Y$
run from $1$ to $t$, and indices in $P$ from $0$ to $p-1$.
\begin{center}\small
\begin{tabular}{ll}
pair&potential\\ \hline
$P$--$P$, $P$--$X$, $X$--$X$ (distinct)&$L$\\
$\{x_i,y_j\}$, $j=i$&$t-i+1$\\
$\{x_i,y_j\}$, $j>i$&$2t-i+1$\\
$\{x_i,y_j\}$, $j<i$&$L-j$\\
$\{x_i,z\}$&$2t-i+1$\\
$\{y_i,y_j\}$, $i<j$&$L-j$\\
$\{y_i,z\}$&$2t+1$\\
$\{p_i,y_j\}$&$\min\{L,\max\{L-j,2t+3+\rho_i\}\}$\\
$\{p_i,z\}$&$\min\{L,2t+2+\rho_i\}$
\end{tabular}
\end{center}
Substitution in the two letter actions gives
\[
 \Phi(E)\le1+\Phi(Ea),\qquad \Phi(E)\le1+\Phi(Eb).
\]
Here are the boundary checks. The last-two-$P$ interchange preserves $\rho$;
a downward $P$-step decreases it by at most one.
For $P$--$Y$ pairs the $a$-boundary $y_t\to z$ decreases the second term
from $2t+3+\rho$ to $2t+2+\rho$.
The $b$-boundary $p_0\to y_1$ with $j<t$ has source $L-j$
and target $L-j-1$, since $j\le t-1$; with $j=t$ its target is at least $L-1$.
For $X$--$Y$, crossing $x_t$ under $a$ gives $Y$--$Y$ with potential $L-j-1$
when $j<i$; crossing under $b$ gives $P$--$Y$ of at least that potential.
The kernel pair $\{x_t,y_t\}$ has potential one.
Crossing $y_t$ in a $Y$--$Y$ pair under $a$ gives $Y$--$z$ of
potential $2t+1=L-t-1$.
For $X$--$z$, the $a$-boundary is $\{x_t,z\}\to\{y_1,x_1\}$,
with values $t+1,t$; for $Y$--$z$, every $a$-image has potential $2t$.
All remaining cases either retain potential, decrement an index by one,
or enter a type whose potential is at least the source minus one.
Clipping at $L$ cannot increase a positive difference.
Thus the inequalities hold for every index, not just tested sizes.
Every distinct pair in $S$ has potential $L$.
Telescoping along a merging word proves its length is at least $L$;
every compression merges an initial pair, giving the claimed lower bound.

\emph{No avoidance before $q$.}
Use singleton $P$-cells and $t$ color cells, giving $x_i,y_i$ color $i$
and $z$ color $1$. Let $T_a$ swap the last two $P$-cells and cycle the colors,
and let $T_b$ cycle $p_0,1,\ldots,t,p_{p-1},\ldots,p_1$.
For the incidence matrix $C$ both letters satisfy
\[
 D_\ell C=CT_\ell+f_z(e_1-e_2).
\]
Here $D_\ell$ is the original-letter transition matrix, $f_z$ the $z$-column, and $e_i$ the color row; the displayed discrepancy is a rank-one outer product.
Since $S$ has one state in every cell, induction along each original word $w$ gives
$\chi_S D_w C=\mathbf1+\sum_j\mu_j(e_1-e_2)T_{w_j}$,
where $\mu_j\ge0$ counts trajectories at $z$ immediately before the corresponding suffix $w_j$.
A trajectory from $X$ needs at least $t+1$ steps to reach $z$;
any entrance from $P$ starts at $y_1$ after at least one step and then needs
at least $t$ more. Thus the first nonzero correction has position at least $t+2$.
In the prescribed cell graph, color $2$ needs $t-1$ letters to enter $P$
at $p_{p-1}$ and at least $p-r$ more to reach $A$.
The only interchange is between adjacent last $P$-points and cannot shorten
this descending path. A negative correction therefore reaches $A$ only at
length at least $(t+2)+(t-1+p-r)=p+2t+1-r=q$.
Before then every protected coordinate has multiplicity at least one.

\emph{Attainment.}
After $a^{2t}b$, the missing $P$-point is $p_{p-1}$; after a further $a$
it is $p_{p-2}$. This is the unique $b^{p-r-1}$-preimage of $p_{r-1}$.
Hence $a^{2t}bab^{p-r-1}$ omits that protected point at length $q$.
The $X\cup Y\cup\{z\}$ portion after $a^{2t}b$ is
$\{x_2,\ldots,x_t,y_1,z\}$. After $a^t$ it contains $x_t$ and $y_t$,
so $a^{2t}ba^tb$ compresses at length $L$.
The existence of $x_t$ in the preceding set uses $t\ge2$.
Combining both upper witnesses with the lower bounds gives $\ac(S,A)=\min\{q,L\}$.

\emph{Structural requirements.}
Put $B=P\cup Y$, $h=p+t$ and $E=B\cup\{z\}$.
Letter $b$ cycles $B$ in the order $p_0,y_1,\ldots,y_t,p_{p-1},\ldots,p_1$,
and $Qb^t=E$. The $a$-cycle connects all other states to $B$, proving
strong connectivity.
On $E$, $d=a^{2t}b^h$ fixes $P$, cycles $y_1\to y_t\to\cdots\to y_1$,
and sends $z$ to $y_t$. Thus $e=d^t$ fixes $B$ and sends $z$ to $y_1$.
Since $Ba^{t+1}=S$, the original-letter word $b^tea^{t+1}$ reaches $S$.
The word $a^{2t+1}$ fixes $X\cup Y\cup\{z\}$ and transposes the last two
$P$-points, adjacent on the $b$-cycle. These words generate every permutation
of $B$, with inverses realized by positive powers.
Moreover $Ba=(B\setminus\{y_1\})\cup\{z\}$,
$Bab=(B\setminus\{y_2\})\cup\{z\}$, and $Babe=B\setminus\{y_2\}$.
Permutations route any pair in a current image into the kernel of $abe$;
repeated compression synchronizes $B$. Prepending $b^te$ synchronizes $Q$.
For $q\le L$, the construction and the universal bound $\ac(S,A)\le q$ determine both maxima.
\end{proof}




\section{Short-pair graphs and the upper bounds}\label{sec:method}
Let $P_h$ join distinct states if some original-letter word of length at most
$h$ merges them. Write $\tau(P_h)$ for its minimum vertex-cover size.
A set survives compression through depth $h$ exactly when it is independent
in $P_h$. Thus
\begin{equation}\label{eq:cover}
 \exists T,\ |T|=n-m,\ c(T)>h
 \quad\Longleftrightarrow\quad \tau(P_h)\le m.
\end{equation}
This is the short-pair viewpoint on subset compression; see also the
computational compression framework in~\cite{kisielewicz2016}.
Figure~\ref{fig:short-pair-cover} illustrates the equivalence.

\begin{lemma}[Safety and union of actual short images]\label{lem:short-images}
Suppose $|S|=n-m\ge2$, $c(S)\ge L$, and $0\le d\le L-2$.
Every $Sw$ with $|w|\le d$ has size $n-m$ and is independent in $P_{L-1-d}$.
In a strongly connected automaton their union has at least
$\min(n,n-m+d)$ states.
\end{lemma}
\begin{proof}
Compression in the prefix $w$, or a merging suffix of length at most
$L-1-d$, would compress $S$ before length $L$.
Let $U_i=\bigcup_{|w|\le i}Sw$. If $U_i$ is proper, strong connectivity
gives an edge from $U_i$ to its complement, whence $|U_{i+1}|\ge|U_i|+1$.
Starting at $|U_0|=n-m$ proves the union bound.
\end{proof}

For an early layer $G=P_h$ and a later layer $P_H$, the reductions retain
the \emph{complete} later maximum-independent-set family $\mathcal D$,
not a chosen sample. Nonisolated core states and every early isolated state
omitted by a row of $\mathcal D$ are named consistently across all rows.
The remaining common isolated states form a reservoir whose possible
entrances and exits are represented by the finite letter signatures.
The support bounds and row-rank restrictions in the supplement make
this necessary domain independent of the original $n$.

Every original letter projects to an admissible signature. The finite
domains enforce the one-step pair-distance inequalities, the existence
of attaining letters, compatible row exits, and necessary strong
connectivity. Where $H-h=2$, the same original letter must also satisfy
the safe-row condition for its square $f^2$.
These constraints hold for every automaton satisfying the proposed
long-compression hypothesis, so excluding all compatible finite domains
proves the corresponding upper bound.

The upper-bound exclusions use the short-pair cover number
to split the finite search. Figure~\ref{fig:coverage} summarizes
the two exact computations.

\subsection{Defect four: excluding twelve steps}\label{sec:m4}
Assume $n\ge11$, $|S|=n-4$, and $c(S)\ge12$.
The known codimension-two bound $C_2\le6$ and
codimension-three bound $C_3\le9$ imply $\tau(P_7)\ge3$ and
$\tau(P_9)\ge4$, whereas \eqref{eq:cover} gives $\tau(P_7)\le4$.
Here $C_j$ denotes the maximum compression distance over all $(n-j)$-sets
in the same automaton. There are exactly two branches.
\begin{enumerate}
\item $\tau(P_7)=4$.
The actual images of depth at most four are maximum independent sets
of $P_7$ and cover $Q$ by Lemma~\ref{lem:short-images}.
The \proofsection{problem4-twelve-seventh-layer}{seven-layer reduction}
bounds the nonisolated core size by $5\le N\le8$, its maximum degree by
four, and its edge count by $7\le e\le10$, with maximum-family row rank
at least five. The complete classification has 91 isomorphism classes.
The complete letter and distance checks leave 152 joint distance vectors,
all excluded by the necessary exit from a proper core.
\item $\tau(P_7)=3$.
Then $\tau(P_9)=4$. The
\proofsection{problem4-twelve-finite-domains}{named-state reduction}
gives 117 early graph classes with $5\le N\le9$ and at most ten named
states in total. The complete later family has three to seven rows,
row rank at least three, and union omitting at most two states.
The closure-compatible families, modulo core automorphisms and consistent
isolated-state renaming, give exactly 10,142 necessary domains.
Every domain is excluded by the common distance, quotient connectivity,
and, where needed, same-letter square conditions.
\end{enumerate}
The proper-core conditions apply because $n\ge11$ exceeds the named support.
The six disjoint computational ranges contain
$1{,}414+259+978+2{,}349+2{,}500+2{,}642=10{,}142$
domains, with no survivors.
Both complete letter-generation orders and both common-distance
implementations were compared in these ranges.
The two branches prove $c(S)\le11$; the full implication chain is in
\proofsection{problem4-four-exact-eleven}{the codimension-four theorem}.
For $n\le10$, $q\le9$ suffices. For all $n$, matching binary witnesses
come from Proposition~\ref{prop:smallq} at $q=5,\ldots,9$ and
Theorem~\ref{thm:two-block} at $q\ge10$.

\subsection{Defect five: excluding fifteen steps}\label{sec:m5}
Assume $n\ge16$, $|S|=n-5$, and $c(S)\ge15$.
Since $C_3\le9$, \eqref{eq:cover} gives $\tau(P_9)\in\{4,5\}$.
The \proofsection{problem4-five-fifteen-domains}{nine-layer reduction}
gives the following exhaustive split.
\begin{enumerate}
\item $\tau(P_9)=5$.
The actual images of depth at most five are maximum independent sets
and cover $Q$. The core has $6\le N\le10$, maximum degree at most five,
at least nine edges, and maximum-family row rank at least six.
Exactly 2,294 classes survive the complete graph classification.
The necessary original-letter defect is at most three, using $C_4\le11$:
defect four would give compression within $2+11<15$.
The row-exit threshold is six and fifth powers remain safe.
The joint distance calculation excludes all 2,294 classes.
Its 72 disjoint ranges are $[32j,32j+31]$ for $0\le j\le70$,
and $[2272,2293]$, so $71\cdot32+22=2294$.
\item $\tau(P_9)=4$.
Take $\mathcal D$ to be the complete maximum-independent-set family
of $P_{11}$. It has row rank at least four and union omitting at most
two states. The named support is at most eleven.
The complete early-core classification has 4,658 classes:
90 with eleven nonisolated states and 4,568 with five to ten.

The following paper argument excludes this remaining domain.
\begin{lemma}[Multipartite overlap exclusion]\label{lem:multipartite}
Let $h\ge1$ and $t\ge2$. Suppose a noncomplete $P_h$ in a synchronizing automaton is a complete multipartite graph on $V=V_1\sqcup\cdots\sqcup V_t$ and has isolated remaining states $J$. Assume each $V_i$ admits a family $\mathcal R_i$ of nonempty subsets whose union is $V_i$, whose intersection graph is connected, and for which every $R\cup J$ with $R\in\mathcal R_i$ is independent in $P_{h+1}$. Then this configuration is impossible.
\end{lemma}
\begin{proof}
A $P_h$ nonedge cannot be merged in one letter. Fix a letter $f$. It is therefore injective on each part and on $J$, and $Jf$ is disjoint from $Vf$. Since every $(R\cup J)f$ is $P_h$-independent, if $Jf\subseteq J$, then $Jf=J$ and $Rf$ is contained in some single part. Intersecting $R,R'$ have intersecting images, so the connected intersection graph forces all of $V_if$ into one part. Every original nonedge remains a nonedge.

Otherwise $Jf$ meets some part $V_j$. Independence of $(R\cup J)f$ forces $Rf\subseteq V_j\cup J$ for every $R$, hence $Vf\subseteq V_j\cup J$ and $Jf\subseteq V_j\cup J$. Again every original nonedge remains a nonedge. This holds for every letter. But a synchronizing automaton with noncomplete $P_h$ must have some letter sending a nonedge to an edge of $P_h$: follow the shortest merging word for a pair at distance greater than $h$ until the distance first drops from $h+1$ to $h$. Contradiction.
\end{proof}
The two pairs of intersecting triples above separately cover the two parts of $K_{4,4}$, and their rows (with common isolates adjoined) are independent in $P_{11}$, hence in $P_{10}$. Lemma~\ref{lem:multipartite} with $h=9$ excludes the only relaxation survivor. Thus the combined complete finite exclusions establish $c(S)\le14$. For $n\le15$, the original bound $q\le14$ suffices. Matching binary witnesses are given by Theorem~\ref{thm:two-block} for $q\ge12$ and Proposition~\ref{prop:smallq}: the first and second cardinality faces cover $q=6,7$, and the cofinal construction covers $q=8,\ldots,11$.

\section{Constructions for the smaller values of \texorpdfstring{$q$}{q}}\label{sec:appendix}
\begin{theorem}\label{thm:sharp-faces}
The following bounds hold, and every lower bound has a binary witness.
\begin{enumerate}
\item $\Max(n,r+1,r)=n-r$ for $n\ge2$ and $1\le r\le n-1$.
\item $\Max(n,r+2,r)=n-r$ for $n\ge4$ and $1\le r\le n-3$.
\item For $r\ge1$, $s-r\ge3$ and $n\ge2s-r-1$, one has
\[
 \Max(n,s,r)=n-r.
\]
\end{enumerate}
For $1\le r<n$, one has $\Max(n,n,r)=1$.
\end{theorem}
Thus the first two cardinality faces are determined, including their full-image boundary. For every fixed $(s,r)$ with $s-r\ge3$, the bound is sharp at every $n\ge2s-r-1$, in particular at $n=r+k(s-r-1)+2$ for $k\ge2$.

We recall the upper bound with a proof, to make the argument self-contained. For a word $w$ let $D_w(q,t)=1$ if $qw=t$, and $0$ otherwise. Then $D_{uv}=D_uD_v$. The characteristic vector $\chi_S$ is a row vector; $\chi_SD_w$ counts trajectories from $S$, with multiplicity, and its support is $Sw$.
\begin{lemma}[Szyku\l a~\cite{szykula2018,szykula2026}]\label{lem:upper}
If $A\nsubseteq Sz$ for some word $z$, then $\ac(S,A)\le n-|A|$.
\end{lemma}
\begin{proof}
Write $r=|A|$, $s=|S|$, and let $V_i$ be the span over $\mathbb R$ of $\chi_SD_w$ for $|w|\le i$. Suppose all words of length at most $n-r$ are safe. Safety gives $|Sw|=|S|$, so $S\to Sw$ is injective: no trajectories collide, and every image state has multiplicity $1$. Their rows therefore have total $s$ and value $1$ at every coordinate of $A$. Hence $V_{n-r}$ lies in
\[
 H=\{x:x_q=s^{-1}\textstyle\sum_{t\in Q}x_t\text{ for every }q\in A\}.
\]
The $r$ equations are independent since $r<s$, so $\dim H=n-r$. Now $\dim V_0=1$, and $V_i=V_{i+1}$ implies invariance under every letter and stabilization forever. Therefore some equality must occur before index $n-r$, or $\dim V_{n-r}\ge n-r+1$. Stabilization puts $\chi_SD_z$ in $H$, contradicting its zero coordinate in $A$ and total $s$.
\end{proof}
In a strongly connected synchronizing automaton with $n\ge2$, a reset word followed by a path to a state outside $A$ gives the required $z$. This proves every upper bound $n-r$ in Theorem~\ref{thm:sharp-faces}. If $S=Q$, some letter must be nonpermutational and immediately compresses $Q$, while the empty word is safe. Thus $\ac(Q,A)=1$.

We will also use the following elementary fact. If $t$ cycles $x_0,x_1,\ldots,x_{s-1}$ and $e$ sends $x_0$ to $x_1$ while fixing the other points, then
\begin{equation}\label{eq:unit}
 e(t^{s-1}e)^{s-2}
\end{equation}
maps this set to $x_1$. Indeed, after $e$ the image is $\{x_1,\ldots,x_{s-1}\}$; each $t^{s-1}e$ deletes the last point of the remaining interval. This also covers $s=2$.

\subsection{The first two cardinality faces}
\begin{proposition}\label{prop:one}
Part~(1) of Theorem~\ref{thm:sharp-faces} holds.
\end{proposition}
\begin{proof}
Put $L=n-r\ge1$ and $Q=\{0,\ldots,n-1\}$. Define
\[
 qa=\begin{cases}q+1&q<L,\\q&q\ge L,\end{cases}
 \qquad S=\{0,L,L+1,\ldots,n-1\},\qquad A=S\setminus\{0\}.
\]
Let $b$ cycle $S$ in the displayed order and send every point outside $S$ to $L$. Then $Qb=S$. The $a$-path from $0$ to $L$ and the $b$-cycle prove strong connectivity.

For $0\le j<L$, set $S_j=A\cup\{j\}$. One has $S_jb=S_0$, and $S_ja=S_{j+1}$ for $j<L-1$; $S_{L-1}a=A$. Induction shows that every word of length less than $L$ has image $S_j$ with $j<L$, so is safe. The word $a^L$ compresses $S$, proving $\ac(S,A)=L$.

On $S$, $c=a^L$ is the unit $0\mapsto L$. Formula~\eqref{eq:unit} with $t=b$ shows that $c(b^rc)^{r-1}$ resets $S$. It resets $Q$ as well, since $Qc=A=Sc$. This proves synchronization, also when $L=1$ or $r=1$.
\end{proof}

\begin{proposition}\label{prop:two}
Part~(2) of Theorem~\ref{thm:sharp-faces} holds.
\end{proposition}
\begin{proof}
Put $M=n-r\ge3$. Take disjoint sets $A=\{p_0,\ldots,p_{r-1}\}$ and $U=\{u_1,\ldots,u_M\}$, and let $S=A\cup\{u_1,u_2\}$. Letter $a$ fixes $A$, while
\[
 p_ib=p_{i+1}\quad(0\le i<r-1),\qquad p_{r-1}b=u_1.
\]
For odd $M$, let $a$ cycle $u_1,\ldots,u_M$ and set
\[
 u_jb=\begin{cases}p_0&j\text{ odd},\\u_2&j\text{ even}.\end{cases}
\]
For even $M$ (necessarily $M\ge4$), set
\[
 u_1a=u_2,\quad u_ja=u_{j+1}\ (2\le j<M),\quad u_Ma=u_2,
 \qquad
 u_jb=\begin{cases}u_2&j=1\text{ or }j\ge3\text{ odd},\\p_0&j\text{ even}.\end{cases}
\]
In both cases $Qb=S$. From $p_0$ the $b$-path reaches $u_1$, whose $a$-orbit covers $U$; some state of $U$ returns to $p_0$ under $b$. Every state therefore reaches and is reached from $p_0$.

The two moving points of $Sa^j$ remain distinct for all $j\ge0$. For $0\le j\le M-2$, their $b$-images are $p_0$ and $u_2$, so $Sa^jb=S$. At $j=M-1$, the pair is $\{u_M,u_1\}$ for odd $M$ and $\{u_M,u_2\}$ for even $M$; both points map to $p_0$. Thus $h=a^{M-1}b$ compresses $S$. Every word shorter than $M$ splits into $a$-runs followed by $b$, each run having length at most $M-2$, and a trailing $a$-run. All these transitions are safe. Hence $\ac(S,A)=M$.

Set $s=r+2$, and $t=ab$ for odd $M$, $t=b$ for even $M$. On $S$, $t$ cycles $p_0,\ldots,p_{r-1},u_1,u_2$. If $e$ is the unit $u_1\mapsto u_2$ on $S$, then $h=et$ there. Consequently
\[
 h(t^{s-2}h)^{s-2}=e(t^{s-1}e)^{s-2}t
\]
on $S$, and this is a reset by~\eqref{eq:unit}. Prepending $b$ gives a reset of $Q$. The unit $e$ is used only to verify the identity; the reset word uses the original letters.
\end{proof}

\subsection{Every larger fixed gap}
\begin{proposition}\label{prop:gaps}
For $r\ge1$, $g\ge2$ and $M\ge2g$, there is a binary strongly connected synchronizing automaton with
\[
 n=r+M+1,\quad |S|=s=r+g+1,\quad |A|=r,
 \qquad \ac(S,A)=M+1.
\]
The subset $S$ is an actual reachable image.
\end{proposition}
\begin{proof}
Take disjoint sets $P=\{p_0,\ldots,p_r\}$ and $U=\{u_1,\ldots,u_M\}$.
Set $A=P\setminus\{p_r\}$ and $S=P\cup\{u_1,\ldots,u_g\}$.
First suppose $M\ge2g+1$ and put $h=1+((M-1)\bmod g)$.
Then $1\le h\le g$ and $M\ge2g+h$.
Let $a$ fix $P$, send $u_j$ to $u_{j+1}$ for $j<M$, and send $u_M$ to $u_{h+1}$.
Define $p_0b=u_h$, $p_ib=p_{i-1}$ for $1\le i\le r$, and
\[
 u_jb=\begin{cases}
 p_r&j\equiv h-1\pmod g,\\
 u_1&j=g,\ h\ge2,\\
 u_{h+g}&j=M,\\
 u_{j+1}&\text{otherwise}.
 \end{cases}
\]
The cases are disjoint. On $S$, $b$ cycles
$p_0,u_h,\ldots,u_g,u_1,\ldots,u_{h-1},p_r,\ldots,p_1$.
The segment $u_1,\ldots,u_{h-1}$ is empty for $h=1$.
Ordinary outside states follow increasing indices to the next index
congruent to $h-1$, then enter $p_r$.
The exceptional path is $u_M,u_{h+g},\ldots,u_{h+2g-1},p_r$, of length $g+1$.
Its indices exist because $M\ge2g+h$.
Thus $Qb^{g+1}=S$. The $a$-path from $u_1$ covers $U$, and the $b$-cycle covers $P$,
proving strong connectivity.

\emph{Safety.}
Use singleton $P$-cells and color cells $1,\ldots,g$, assigning
$u_j$ color $1+((j-h-1)\bmod g)$.
Let $C$ be the $n\times s$ state-to-cell incidence matrix.
Let $T_a$ fix $P$ and cycle the colors, and let $T_b$ cycle
$p_0,g,1,\ldots,g-1,p_r,\ldots,p_1$.
For a standard state column $f_q$ and cell row $e_c$, one has
\begin{equation}\label{eq:error}
 D_aC=CT_a,\qquad D_bC=CT_b+f_{u_M}(e_g-e_1).
\end{equation}
The last term is an outer product of an $n\times1$ column and a $1\times s$ row.
These cells are bookkeeping devices, not a quotient automaton.
For $w=\ell_1\cdots\ell_t$, iteration gives
\begin{equation}\label{eq:expand}
 \chi_SD_wC=\one_s+
 \sum_{\substack{1\le i\le t\\\ell_i=b}}
 \mu_i(e_g-e_1)T_{\ell_{i+1}\cdots\ell_t},
 \qquad \mu_i=(\chi_SD_{\ell_1\cdots\ell_{i-1}})_{u_M}\ge0.
\end{equation}
This counts trajectories even after collisions, with products in word order.
A trajectory from $S$ needs at least $M-g$ letters to reach $u_M$:
initial indices are at most $g$, every transition within $U$ increases an
index by at most one, and entrances from $P$ have index $h\le g$.
Hence a nonzero correction has $i\ge M-g+1$.
In the cell graph, color $g$ needs at least $g-1$ letters to reach color $g-1$,
and color $1$ needs at least $g$ letters to reach $A$.
The latter route traverses $1,\ldots,g-1$ and then takes two letters $b$
through $p_r$. A detour through $P$ cannot shorten either route.

For $t\le M-1$, correction suffixes have length at most $g-2$,
so no positive correction reaches color $g-1$, whose trajectory count is at most one.
Suppose a word of length at most $M$ has a first collision.
For a colliding $b$, the distinct source states must lie in its only
non-singleton fiber, all of color $g-1$, contradicting this count bound.
For a colliding $a$, the sources must be $u_h,u_M$, its only non-singleton fiber.
They are not both initially present. A preceding $b$ cannot give $u_M$,
which has no $b$-preimage. A preceding $a$ requires sources
$u_{h-1},u_{M-1}$ one step earlier, both of color $g-1$, again a contradiction;
for $h=1$, $u_h$ has no $a$-preimage at all.
Thus every word of length at most $M$ preserves $s$ states.
For $t\le M$, every correction suffix has length at most $g-1$,
so no negative correction reaches $A$. Its singleton cells retain multiplicity
at least one. Every such word is safe.

The word $a^{M-g}b^{g+1}$ has length $M+1$.
Exactly its first $b$ has a trajectory at $u_M$, with multiplicity one.
Its negative correction is $e_1T_b^g=e_{p_{r-1}}$, whereas
its positive correction is $e_gT_b^g=e_{p_r}$.
It therefore omits $p_{r-1}$ and attains equality, also when $r=1$.

For synchronization put $c=a^{M-h}b^s$.
The $a$-run fixes the cyclic tail $u_{h+1},\ldots,u_M$.
For $j<h$, it sends $u_j$ to $u_{M-h+j}$, which has the same remaining
$b$-distance $h-j$ to $p_r$ as $u_j$.
For $j=h$, it sends $u_h$ to $u_M$, which reaches $p_0$ in $g+1+r=s$ letters $b$.
Consequently $c$ fixes $S\setminus\{u_h\}$ and sends $u_h$ to its $b$-predecessor $p_0$.
After $c$, each $bc$ extends the missing interval by one point in forward
$b$-order. Thus $b^{g+1}c(bc)^{s-2}$ resets $Q$.

\emph{The boundary $M=2g$.}
Let $a$ fix $P$ and cycle $U$.
Define $p_0b=u_g$, $p_ib=p_{i-1}$ and
\[
 u_jb=\begin{cases}
 p_r&j\equiv g-1\pmod g,\\
 u_{j+1}&j<M,\ j\not\equiv g-1\pmod g,\\
 u_M&j=M.
 \end{cases}
\]
Now $b$ cycles $B=P\cup\{u_g,\ldots,u_{2g-1}\}$ and fixes $u_{2g}$;
$Qb^{g+1}=E=B\cup\{u_{2g}\}$.
Put $d=a^{g-1}b^s$. It fixes $P$ and on $E$ has actions
\[
 u_gd=u_{2g-1},\quad u_{g+1}d=u_{2g},\quad
 u_{g+j}d=u_{g+j-1}\ (2\le j\le g-1),\quad u_{2g}d=u_{2g-1}.
\]
Therefore $Eda^g=S$, proving actual reachability. The two cycles prove strong connectivity.
Assigning $u_j$ color $1+((j-1)\bmod g)$ gives exactly
\eqref{eq:error}--\eqref{eq:expand} again.
The same delay and collision arguments apply, with $a$ now a permutation.
The word $a^gb^{g+1}$ attains avoidance: only the first negative correction
has a suffix of $g$ letters $b$ and reaches $p_{r-1}$; no positive correction reaches $A$.
Finally $e=d^g$ fixes $E\setminus\{u_g\}$ and sends $u_g$ to $u_{2g}$.
Each $be$ removes one further point of $B$ while retaining $u_{2g}$.
Hence $b^{g+1}e(be)^{s-1}$ resets $Q$.
\end{proof}
Taking $g=s-r-1$ and $M=n-r-1$ proves part~(3) of Theorem~\ref{thm:sharp-faces}.
